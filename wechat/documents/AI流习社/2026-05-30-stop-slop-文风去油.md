import re
import random
import sys

# 声明式高频 AI 口癖黑名单及人性化多维同义词代换库，对于中文不使用 \b 边界断言，拒绝任何占位
SLOP_DICTIONARY = {
    r"\bdelve into\b": ["explore", "examine", "look closely at", "investigate"],
    r"\btestament to\b": ["proof of", "sign of", "shows how", "demonstrates"],
    r"\bmoreover\b": ["also", "in addition", "what's more", ""],
    r"\bfurthermore\b": ["plus", "besides", "additionally", ""],
    r"\bnotwithstanding\b": ["despite", "even with", "in spite of"],
    r"双刃剑": ["利弊并存", "硬币的两面", "一把双面刃"],
    r"不得不说": ["老实说", "坦白讲", "客观来看"],
    r"令人瞩目": ["亮眼", "出色", "引起关注"]
}

class TextSlopScanner:
    def __init__(self):
        # 将字符串正则编译成高性能模式对象
        self.compile_rules = {re.compile(pattern, re.IGNORECASE): pattern for pattern in SLOP_DICTIONARY.keys()}

    def scan_slop_density(self, raw_text):
        """工作流一：静态分析文本并匹配高频 AI 口癖，计算文本含油量"""
        findings = []
        total_words = len(raw_text.split()) if len(raw_text.split()) > 0 else len(raw_text)
        
        for regex, raw_pattern in self.compile_rules.items():
            matches = regex.findall(raw_text)
            if matches:
                findings.append({
                    "pattern": raw_pattern,
                    "count": len(matches),
                    "instances": list(set(matches))
                })
                
        total_slop_count = sum(item["count"] for item in findings)
        slop_density = (total_slop_count / total_words) * 100.0 if total_words > 0 else 0.0
        
        return findings, slop_density


class TextDecounterBreaker:
    def __init__(self):
        self.compile_rules = {re.compile(pattern, re.IGNORECASE): replacements for pattern, replacements in SLOP_DICTIONARY.items()}

    def de_slop_refine(self, raw_text):
        """工作流二：动态语义代换并融合多样性同义词降噪重组，去除文风油腻感"""
        refined_text = raw_text
        
        # 1. 执行动态替换
        for regex, replacements in self.compile_rules.items():
            def replace_callback(match):
                # 随机选择同义词，确保每次生成的文笔具有多样性与人类的随机质感
                replacement = random.choice(replacements)
                # 保持首字母大小写一致性
                if match.group(0)[0].isupper() and len(replacement) > 0:
                    return replacement[0].upper() + replacement[1:]
                return replacement
            
            refined_text = regex.sub(replace_callback, refined_text)
            
        # 2. 物理格式美化：合并因删除词汇导致的多余空行和前后多余空格
        refined_text = re.sub(r"\s+", " ", refined_text)
        refined_text = re.sub(r" ([,.!?])", r"\1", refined_text) # 修正标点空格
        
        return refined_text.strip()


# ==================== 仿真运行与测试驱动入口 ====================
if __name__ == "__main__":
    print("[⚙] 正在初始化 stop-slop 赛博文风去油与 NLP 降噪重组双引擎...")
    
    scanner = TextSlopScanner()
    refiner = TextDecounterBreaker()

    # 模拟一段带有极其明显、浓郁 AI 阶段与口癖特征的原始草稿文本
    mock_ai_text = (
        "Moreover, we must delve into this technology. "
        "It is a testament to our continuous hard work. "
        "Furthermore, this tool is a 双刃剑, but 不得不说, its performance is 令人瞩目."
    )

    print("\n--------------------------------------------------")
    print("[演示一：静态 AI 特征口癖深度扫描]")
    findings, density = scanner.scan_slop_density(mock_ai_text)
    print(f" 📂 原始草稿字数: {len(mock_ai_text.split())} 词")
    print(f" 📈 AI 特征口癖密度: {density:.2f}%")
    print(" 🔍 命中的 AI 垃圾废话特征详情:")
    for finding in findings:
        print(f"  -> 词型 [{finding['pattern']}] 共匹配到 {finding['count']} 次")

    print("\n--------------------------------------------------")
    print("[演示二：动态同义词代换与文风降噪提纯]")
    refined_result = refiner.de_slop_refine(mock_ai_text)
    
    print("\n[📊 去油成功] 最终呈献给用户的清爽、高质人类黄金文本：\n")
    print(f" 💚 提纯文本: \"{refined_result}\"\n")

    # 验证是否成功物理剥离了 high-frequency AI tells，且语义代换逻辑 100% 成功
    ai_tells_gone = (
        "delve into" not in refined_result.lower()
        and "testament to" not in refined_result.lower()
        and "双刃剑" not in refined_result
        and "不得不说" not in refined_result
    )
    if ai_tells_gone:
        print("[✔ 引擎测试结论] AI口癖静态清洗与语义多样性重组去油 100% 成功！")
        sys.exit(0)
    else:
        print("[❌ 致命错误] 去油不彻底，AI特征口癖依然残留泄漏！")
        sys.exit(1)
