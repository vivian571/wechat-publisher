### <font color='Purple'><b>2. 核心引擎：AI视频生成器</b></font>

接下来，我们创建AI视频生成的核心引擎：

```python
import torch
from diffusers import DiffusionPipeline, DPMSolverMultistepScheduler
from diffusers.utils import export_to_video
from transformers import AutoTokenizer, AutoModel
import numpy as np
import os
from datetime import datetime
from moviepy.editor import VideoFileClip, concatenate_videoclips

class AIVideoGenerator:
    def __init__(self):
        # 初始化模型
        self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self.pipe = DiffusionPipeline.from_pretrained(
            "damo-vilab/text-to-video-ms-1.7b", 
            torch_dtype=torch.float16,
            variant="fp16"
        )
        self.pipe.scheduler = DPMSolverMultistepScheduler.from_config(self.pipe.scheduler.config)
        self.pipe.to(self.device)
        
        # 初始化中英文翻译模型
        self.tokenizer = AutoTokenizer.from_pretrained("Helsinki-NLP/opus-mt-zh-en")
        self.model = AutoModel.from_pretrained("Helsinki-NLP/opus-mt-zh-en")
        
        # 创建输出目录
        self.output_dir = "ai_videos"
        os.makedirs(self.output_dir, exist_ok=True)
        
        print("✅ AI视频生成器初始化完成！")
    
    def translate_zh_to_en(self, text):
        """将中文翻译为英文，提高生成质量"""
        inputs = self.tokenizer(text, return_tensors="pt")
        outputs = self.model.generate(**inputs)
        translation = self.tokenizer.decode(outputs[0], skip_special_tokens=True)
        return translation
    
    def enhance_prompt(self, prompt):
        """增强提示词，提高视频质量"""
        quality_terms = ["high quality", "4K resolution", "cinematic", "professional"]
        style_terms = ["movie style", "Hollywood quality", "professional production"]
        
        enhanced = prompt + ", " + ", ".join(quality_terms) + ", " + ", ".join(style_terms)
        return enhanced
    
    def text_to_video(self, prompt_zh, duration=10, fps=8):
        """从文本生成视频"""
        # 翻译提示词
        prompt_en = self.translate_zh_to_en(prompt_zh)
        # 增强提示词
        prompt_en = self.enhance_prompt(prompt_en)
        
        print(f"🔄 正在生成视频: {prompt_zh}")
        print(f"🔄 翻译后的提示词: {prompt_en}")
        
        # 计算需要生成的帧数
        num_frames = duration * fps
        
        # 生成视频
        video_frames = self.pipe(prompt_en, num_inference_steps=50, num_frames=num_frames).frames
        
        # 保存视频
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        video_path = os.path.join(self.output_dir, f"video_{timestamp}.mp4")
        export_to_video(video_frames, video_path, fps=fps)
        
        print(f"✅ 视频生成完成: {video_path}")
        return video_path
    
    def combine_videos(self, video_paths, output_path=None):
        """合并多个视频片段"""
        if not output_path:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            output_path = os.path.join(self.output_dir, f"combined_{timestamp}.mp4")
        
        clips = [VideoFileClip(path) for path in video_paths]
        final_clip = concatenate_videoclips(clips)
        final_clip.write_videofile(output_path)
        
        print(f"✅ 视频合并完成: {output_path}")
        return output_path
```

<font color='Red'><b>这个类封装了所有AI视频生成的核心功能，包括模型加载、中英文翻译、提示词增强和视频生成！</b></font>

# 批量生成和场景拼接
def batch_generate_and_combine():
    print("===== 批量生成视频并拼接 =====\n")
    
    # 收集场景描述
    scenes = []
    while True:
        scene = input(f"请输入第{len(scenes)+1}个场景描述（输入'完成'结束）：")
        if scene.lower() in ['完成', 'done', 'finish', 'exit']:
            break
        duration = input(f"场景{len(scenes)+1}的时长（秒）[默认5秒]：")
        duration = int(duration) if duration.strip() else 5
        scenes.append((scene, duration))
    
    if not scenes:
        print("没有输入任何场景，退出程序")
        return
    
    # 初始化生成器
    generator = AIVideoGenerator()
    
    # 生成每个场景的视频
    video_paths = []
    for i, (scene, duration) in enumerate(scenes):
        print(f"\n正在生成第{i+1}个场景: {scene}")
        video_path = generator.text_to_video(scene, duration=duration)
        video_paths.append(video_path)
    
    # 合并所有场景
    combined_path = generator.combine_videos(video_paths)
    
    print(f"\n✅ 所有场景已合并！最终视频保存在: {combined_path}")
    print("\n🎬 是否播放合并后的视频？(y/n)")
    if input().lower() == 'y':
        os.system(f"start {combined_path}")

if __name__ == "__main__":
    batch_generate_and_combine()
```

<font color='Orange'><b>这个功能让你可以创建多个场景，然后自动拼接成一个完整故事！</b></font>

# 视频后期处理
from moviepy.editor import VideoFileClip, TextClip, CompositeVideoClip
from moviepy.video.fx import all as vfx

def enhance_video(video_path, output_path=None):
    if not output_path:
        output_path = video_path.replace('.mp4', '_enhanced.mp4')
    
    # 加载视频
    clip = VideoFileClip(video_path)
    
    # 增强亮度和对比度
    clip = clip.fx(vfx.colorx, 1.2)  # 增加亮度
    
    # 添加文字
    txt = TextClip("AI生成视频", fontsize=30, color='white')
    txt = txt.set_position(('right', 'bottom')).set_duration(clip.duration)
    
    # 合成最终视频
    final = CompositeVideoClip([clip, txt])
    
    # 保存
    final.write_videofile(output_path)
    return output_path
```
